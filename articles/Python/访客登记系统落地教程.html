# 访客登记系统落地教程

目标：门卫网页提交表单 → 先落本地 SQLite 兜底 → 自动推送钉钉群通知 → 自动同步到钉钉 AI 多维表格 同时配套：命令行 CLI 操作、MCP 给 AI 工具（WorkBuddy / Cursor）调用 架构原则：本地 SQLite 是唯一可信数据源，钉钉消息、钉钉表格只是同步视图 适用：厂区访客登记，后续可以无缝切换成 MySQL

# 一、整体技术栈

  • 后端：FastAPI（接口服务）
  • 数据库：SQLite（单文件，不用单独装服务，开发/单机部署首选）
  • CLI：Typer
  • MCP：FastMCP
  • HTTP 请求：httpx
  • ORM：SQLAlchemy
  • 能力：钉钉群通知、钉钉 Bitable 多维表格同步、失败重试补偿

# 二、前置准备（钉钉侧，必须先做完）

# 2.1 钉钉开放平台自建应用

  1. 打开钉钉开放平台，创建企业内部自建应用
  2. 拿到 appkey、appsecret
  3. 申请权限：
    • chat:bot / 群机器人（发消息，二选一）
    • bitable:app 多维表格读写权限
  4. （方案 A 简单）群机器人 webhook：钉钉群 → 机器人 → 添加自定义机器人，拿到 webhook 地址，适合快速发通知
  5. （方案 B Bitable）新建钉钉 AI 表格（多维表格）
    • 复制 app_token（表格 ID）、table_id（数据表 ID）
    • 打开表格字段设置，拿到每一列的 field_id（重点！API 不认中文名字）
      • 姓名字段、手机号、车牌、拜访对象、来访事由

记录下来下面这些配置，后面代码要用

DING_APP_KEY = ""
DING_APP_SECRET = ""
BITABLE_APP_TOKEN = ""
BITABLE_TABLE_ID = ""
# 字段ID映射
FIELD_NAME = "fldxxxx"
FIELD_PHONE = "fldyyyy"
FIELD_CAR = "fldzzzz"
FIELD_VISIT_TO = "fldaaaa"
FIELD_REASON = "fldbbbb"

# 四、代码文件

# 4.1 models.py 数据库表模型

from sqlalchemy import create_engine, Column, Integer, String, Text, DateTime
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import sessionmaker
from datetime import datetime

# sqlite数据库文件，直接在当前目录生成erp.db
SQLITE_URL = "sqlite:///./erp.db"
engine = create_engine(SQLITE_URL, connect_args={"check_same_thread": False})
SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine)
Base = declarative_base()

class Visitor(Base):
    __tablename__ = "visitor"
    id = Column(Integer, primary_key=True, index=True)
    name = Column(String(50), nullable=False)
    phone = Column(String(20), nullable=False)
    car = Column(String(30), default="")
    visit_to = Column(String(50), default="")
    reason = Column(Text, default="")
    created_at = Column(DateTime, default=datetime.now)
    # 同步状态标记，用来重试
    ding_notify_status = Column(String(20), default="pending") # pending/success/fail
    ding_table_status = Column(String(20), default="pending")
    ding_table_record_id = Column(String(100), default="")

# 创建表，第一次运行自动建表
Base.metadata.create_all(bind=engine)

# 获取数据库会话
def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()

# 4.2 service.py 公共业务层（最重要！后端、cli、mcp 全部复用）

import httpx
from models import Visitor, SessionLocal
from datetime import datetime

# ===================== 钉钉配置区，改成你自己的 =====================
DING_WEBHOOK = "https://oapi.dingtalk.com/robot/send?access_token=xxx"
BITABLE_APP_TOKEN = "bxxxxxxxx"
BITABLE_TABLE_ID = "tblxxxxxxxx"
FIELD_NAME = "fldxxxx"
FIELD_PHONE = "fldyyyy"
FIELD_CAR = "fldzzzz"
FIELD_VISIT_TO = "fldaaaa"
FIELD_REASON = "fldbbbb"
DING_APP_KEY = ""
DING_APP_SECRET = ""
# ==================================================================

async def send_ding_msg(content: str) -> dict:
    """发送钉钉群机器人消息"""
    payload = {
        "msgtype": "text",
        "text": {"content": content}
    }
    try:
        async with httpx.AsyncClient(timeout=10) as c:
            resp = await c.post(DING_WEBHOOK, json=payload)
            return resp.json()
    except Exception as e:
        return {"success": False, "msg": str(e)}

async def add_bitable_record(visitor: Visitor) -> dict:
    """写入钉钉多维表格Bitable"""
    # 如果你用自建应用方式，这里先获取access_token
    # 这里演示webhook机器人场景，如果你是自建应用，替换成官方bitable接口
    token_url = f"https://oapi.dingtalk.com/gettoken?appkey={DING_APP_KEY}&appsecret={DING_APP_SECRET}"
    async with httpx.AsyncClient() as c:
        token_resp = await c.get(token_url)
        access_token = token_resp.json()["access_token"]
    url = f"https://oapi.dingtalk.com/v1.0/bitable/apps/{BITABLE_APP_TOKEN}/tables/{BITABLE_TABLE_ID}/records?access_token={access_token}"
    payload = {
        "fields": {
            FIELD_NAME: visitor.name,
            FIELD_PHONE: visitor.phone,
            FIELD_CAR: visitor.car,
            FIELD_VISIT_TO: visitor.visit_to,
            FIELD_REASON: visitor.reason
        }
    }
    try:
        async with httpx.AsyncClient(timeout=10) as c:
            res = await c.post(url, json=payload)
            return res.json()
    except Exception as e:
        return {"success": False, "msg": str(e)}

async def create_visitor(name: str, phone: str, car: str = "", visit_to: str = "", reason: str = "") -> dict:
    """新增访客【核心业务函数】，后端/CLI/MCP共用"""
    db = SessionLocal()
    try:
        # 1. 先写入本地数据库（兜底，第一步！）
        new_v = Visitor(
            name=name,
            phone=phone,
            car=car,
            visit_to=visit_to,
            reason=reason
        )
        db.add(new_v)
        db.commit()
        db.refresh(new_v)

        # 2. 组装通知消息
        ding_text = f"""【访客登记】
姓名：{name}
手机号：{phone}
车牌：{car}
拜访对象：{visit_to}
事由：{reason}
登记时间：{datetime.now().strftime('%Y-%m-%d %H:%M:%S')}"""

        # 3. 推送钉钉消息
        ding_ret = await send_ding_msg(ding_text)
        if ding_ret.get("errcode") == 0:
            new_v.ding_notify_status = "success"
        else:
            new_v.ding_notify_status = "fail"

        # 4. 同步到钉钉多维表格
        bitable_ret = await add_bitable_record(new_v)
        if bitable_ret.get("code") == 0:
            new_v.ding_table_status = "success"
            new_v.ding_table_record_id = bitable_ret.get("record_id", "")
        else:
            new_v.ding_table_status = "fail"
        db.commit()

        return {
            "success": True,
            "visitor_id": new_v.id,
            "ding_notify": ding_ret,
            "bitable": bitable_ret
        }
    except Exception as e:
        db.rollback()
        return {"success": False, "msg": str(e)}
    finally:
        db.close()

# 4.3 main.py FastAPI 后端（网页表单提交接口）

from fastapi import FastAPI, Depends
from pydantic import BaseModel
from service import create_visitor
from fastapi.middleware.cors import CORSMiddleware

app = FastAPI(title="访客登记后端")
# 允许前端页面跨域
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_methods=["*"],
    allow_headers=["*"],
)

class VisitorForm(BaseModel):
    name: str
    phone: str
    car: str = ""
    visit_to: str = ""
    reason: str = ""

@app.post("/api/visitor")
async def add_visitor(form: VisitorForm):
    res = await create_visitor(
        name=form.name,
        phone=form.phone,
        car=form.car,
        visit_to=form.visit_to,
        reason=form.reason
    )
    return res

if __name__ == "__main__":
    import uvicorn
    uvicorn.run("main:app", host="0.0.0.0", port=8080, reload=True)

# 4.4 cli.py Typer 命令行工具（运维手动调用）

import typer
import asyncio
from service import create_visitor, send_ding_msg

app = typer.Typer()

@app.command("visitor:add")
def visitor_add(
    name: str = typer.Option(..., help="访客姓名"),
    phone: str = typer.Option(..., help="手机号"),
    car: str = typer.Option("", help="车牌"),
    visit_to: str = typer.Option("", help="拜访对象"),
    reason: str = typer.Option("", help="来访事由"),
):
    """命令行新增访客记录"""
    ret = asyncio.run(create_visitor(name, phone, car, visit_to, reason))
    if ret["success"]:
        typer.echo(f"✅ 登记成功，访客ID：{ret['visitor_id']}")
    else:
        typer.echo(f"❌ 失败：{ret['msg']}", err=True)

@app.command("ding:send")
def ding_send(content: str):
    """单独发钉钉消息"""
    ret = asyncio.run(send_ding_msg(content))
    print(ret)

@app.command("mcp")
def run_mcp():
    """启动MCP服务"""
    import subprocess
    subprocess.run(["python", "mcp_server.py"])

if __name__ == "__main__":
    app()

# 4.5 mcp_server.py MCP 服务，给 WorkBuddy / Cursor 调用

from fastmcp import FastMCP
from service import create_visitor, send_ding_msg

mcp = FastMCP("访客登记系统")

@mcp.tool()
async def add_visitor(name: str, phone: str, car: str = "", visit_to: str = "", reason: str = "") -> dict:
    """新增访客登记，自动入库、发钉钉通知、同步钉钉多维表格
    Args:
        name: 访客姓名
        phone: 访客手机号
        car: 车牌号码
        visit_to: 拜访的内部人员
        reason: 来访事由
    """
    return await create_visitor(name, phone, car, visit_to, reason)

@mcp.tool()
async def send_ding(content: str) -> dict:
    """发送钉钉群通知消息"""
    return await send_ding_msg(content)

if __name__ == "__main__":
    mcp.run(transport="stdio")

# 六、补偿重试机制（重要补充，解决钉钉接口偶尔失败）

当前代码只是提交时同步一次；如果钉钉接口超时，状态标记为 fail。 新增一个定时脚本，每隔几分钟扫描数据库里失败的记录，重试同步钉钉消息、表格。 这个是生产稳定性关键，这里只给思路：

  1. 写一个 retry_task.py，查询 ding_notify_status="fail"、ding_table_status="fail" 的记录
  2. 循环重新调用 send_ding_msg、add_bitable_record，成功之后更新状态
  3. Windows 用任务计划、Linux 用 crontab 定时跑

# 七、选型 & 扩容说明

  1. 单机厂区、并发不大：保持 SQLite，不用改
  2. 多门卫终端、并发高：只需要修改 models 里面的数据库连接字符串，切换 MySQL，业务代码一行不用改
  3. 权限区分：
    • 网页表单：走 HTTP 接口，业务主入口
    • CLI：运维本地人工操作
    • MCP：AI Agent 自动调用
  4. 数据底线：erp.db 永远是原始数据，钉钉表格只是展示视图

# 八、常见坑清单

  1. Bitable 接口必须使用 field_id，不能用中文列名
  2. 钉钉 access_token 有 2 小时有效期，正式版一定要加内存缓存，不要每次请求都拿 token
  3. MCP 模式里面不要随便 print 日志，会破坏 MCP stdio 协议报文
  4. SQLite 并发弱，如果同时 5 个以上门卫提交，建议升级 MySQL
  5. 钉钉开放平台 Bitable 有 QPS 限流，大量提交不要短时间狂刷
